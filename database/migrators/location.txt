-- Locations
CREATE TABLE location (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    region_id INTEGER NOT NULL,
    field_id INTEGER, -- For field spawns
    terrain_id INTEGER, -- For general terrain
    description TEXT,


    -- Maybe change from rectangle to polygon?
    x_coordinate FLOAT,
    y_coordinate FLOAT,
    height FLOAT,
    width FLOAT, 

    travel_time INTEGER, -- rating for how long it takes to travel across the location

    parent_location_id INTEGER, -- For sub-zones or nested areas
    accessibility INTEGER DEFAULT 1, -- How accessible the location is (1-5 scale)
    FOREIGN KEY (region_id) REFERENCES region(id),
    FOREIGN KEY (parent_location_id) REFERENCES location(id)
);

CREATE TABLE location_pokemon {
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    location_id INTEGER NOT NULL,
    pokemon_id INTEGER NOT NULL,
    rate INTEGER NOT NULL, -- relative spawn rate or weight
    time_of_day TEXT CHECK (time_of_day IN ('day', 'night', 'dawn', 'dusk', 'any')) DEFAULT 'any',
    season TEXT CHECK (season IN ('spring', 'summer', 'autumn', 'winter', 'any')) DEFAULT 'any',
    min_level INTEGER DEFAULT 1,
    max_level INTEGER DEFAULT 100,
    start_date TEXT,
    end_date TEXT,
    encounter_type TEXT CHECK(encounter_type IN ('grass', 'water', 'cave', 'sky', 'fishing', 'event')) DEFAULT 'grass',
    FOREIGN KEY (location_id) REFERENCES location(id)
}
